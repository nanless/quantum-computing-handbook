cmake_minimum_required(VERSION 3.20)
project(HandbookParallelRecovery LANGUAGES CXX)
if(NOT SQD_SOURCE OR NOT RANDOM123_SOURCE)
 message(FATAL_ERROR "Set full SQD_SOURCE and RANDOM123_SOURCE checkouts")
endif()
add_subdirectory(${SQD_SOURCE} upstream EXCLUDE_FROM_ALL)
add_executable(parallel_recovery parallel_recovery.cpp)
target_compile_features(parallel_recovery PRIVATE cxx_std_17)
target_include_directories(parallel_recovery PRIVATE ${SQD_SOURCE}/include ${RANDOM123_SOURCE}/include)
target_link_libraries(parallel_recovery PRIVATE boost_dynamic_bitset bitset2)
if(SQD_ENABLE_OPENMP)
 find_package(OpenMP REQUIRED)
 target_link_libraries(parallel_recovery PRIVATE OpenMP::OpenMP_CXX)
endif()
