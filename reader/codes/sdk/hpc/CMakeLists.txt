cmake_minimum_required(VERSION 3.20)
project(HandbookSQDValidation LANGUAGES CXX)
if(NOT SQD_SOURCE)
  message(FATAL_ERROR "Set SQD_SOURCE to the complete pinned SQD-HPC checkout")
endif()
add_subdirectory(${SQD_SOURCE} upstream EXCLUDE_FROM_ALL)
add_executable(handbook_sqd validate_hpc.cpp)
target_compile_features(handbook_sqd PRIVATE cxx_std_17)
target_include_directories(handbook_sqd PRIVATE ${SQD_SOURCE}/include)
target_link_libraries(handbook_sqd PRIVATE boost_dynamic_bitset bitset2)
